INTERNAL AI CONCEPT

Internal Data Intelligence Assistant

A private, governed AI experience that allows stakeholders to ask questions, understand Tableau dashboards, query approved Databricks data, generate visual insights and escalate unresolved questions to the correct human team.

THE IDEA

What if stakeholders could simply ask the data?

Today, stakeholders often depend on Data and Analytics teams to understand dashboards, interpret KPIs, retrieve information or investigate changes. This project explores a secure AI-powered experience where stakeholders can ask those questions directly using natural language and receive trusted answers from approved enterprise data.

def format_chart(
    fig,
    height=480,
    x_title=None,
    y_title=None
):
    """
    Applies common dashboard formatting to Plotly charts.
    """

    fig.update_layout(
        height=height,
        margin=dict(
            l=40,
            r=30,
            t=80,
            b=50
        ),
        title=dict(
            font=dict(size=21)
        ),
        legend_title_text="",
        hoverlabel=dict(
            font_size=13
        )
    )

    if x_title is not None:
        fig.update_xaxes(
            title_text=x_title
        )

    if y_title is not None:
        fig.update_yaxes(
            title_text=y_title
        )

    return fig

biggest_clusters_pd = spark.sql(f"""
WITH minute_usage AS (

    SELECT
        workspace_id,
        cluster_id,
        usage_minute,

        COUNT(DISTINCT instance_id)
            AS running_workers

    FROM {NODE_USAGE_VIEW}

    WHERE driver = FALSE

      AND start_time >=
          CURRENT_TIMESTAMP() - INTERVAL {LOOKBACK_DAYS} DAYS

    GROUP BY
        workspace_id,
        cluster_id,
        usage_minute
),

cluster_peaks AS (

    SELECT
        workspace_id,
        cluster_id,

        MAX(running_workers)
            AS peak_workers

    FROM minute_usage

    GROUP BY
        workspace_id,
        cluster_id
)

SELECT
    CAST(p.workspace_id AS STRING)
        AS workspace_id,

    p.cluster_id,

    COALESCE(
        i.cluster_name,
        p.cluster_id
    ) AS cluster_name,

    i.cluster_family,

    p.peak_workers

FROM cluster_peaks p

LEFT JOIN {INVENTORY_VIEW} i

    ON p.workspace_id = i.workspace_id
    AND p.cluster_id = i.cluster_id

ORDER BY peak_workers DESC

LIMIT 5
""").toPandas()

fig = px.bar(

quota_pd = spark.sql(f"""
WITH observed_usage AS (

    SELECT
        usage_minute,
        workspace_id,

        COUNT(DISTINCT instance_id)
            AS concurrent_workers

    FROM {NODE_USAGE_VIEW}

    WHERE driver = FALSE

      AND start_time >=
          CURRENT_TIMESTAMP() - INTERVAL {LOOKBACK_DAYS} DAYS

    GROUP BY
        usage_minute,
        workspace_id
),

peak_usage AS (

    SELECT
        workspace_id,

        MAX(concurrent_workers)
            AS peak_workers

    FROM observed_usage

    GROUP BY workspace_id
)

SELECT
    CAST(workspace_id AS STRING)
        AS workspace_id,

    peak_workers,

    {QUOTA_LIMIT}
        AS quota_limit,

    ROUND(
        peak_workers / {float(QUOTA_LIMIT)} * 100,
        2
    ) AS quota_usage_pct

FROM peak_usage

ORDER BY quota_usage_pct DESC
""").toPandas()

def get_capacity_status(value):

    if value >= 90:
        return "CRITICAL"

    if value >= 75:
        return "WARNING"

    return "NORMAL"

quota_pd["status"] = (
    quota_pd["quota_usage_pct"]
    .apply(get_capacity_status)
)

display(quota_pd)

avg_job_runtime = spark.sql(f"""
SELECT
    ROUND(
        AVG(run_duration_minutes),
        2
    ) AS value

FROM {JOB_RUN_VIEW}

WHERE run_start >=
      CURRENT_TIMESTAMP() - INTERVAL {LOOKBACK_DAYS} DAYS
""").first()["value"]


avg_daily_workers = spark.sql(f"""
WITH daily AS (

    SELECT
        usage_date,

        COUNT(DISTINCT instance_id)
            AS workers

    FROM {NODE_USAGE_VIEW}

    WHERE driver = FALSE

      AND start_time >=
          CURRENT_TIMESTAMP() - INTERVAL {LOOKBACK_DAYS} DAYS

    GROUP BY usage_date
)

SELECT
    ROUND(
        AVG(workers),
        2
    ) AS value

FROM daily
""").first()["value"]


avg_job_text = (
    f"{float(avg_job_runtime):,.2f} min"
    if avg_job_runtime is not None
    else "N/A"
)

    f"{float(avg_daily_workers):,.2f}"
